cmake_minimum_required(VERSION 3.16)
project(trac_ik VERSION 2.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(TRAC_IK_BUILD_PYTHON "Build Python 3 bindings" ON)
option(TRAC_IK_BUILD_EXAMPLES "Build standalone IK example" ON)
add_subdirectory(trac_ik_lib)
if(TRAC_IK_BUILD_PYTHON)
  add_subdirectory(trac_ik_python)
endif()
if(TRAC_IK_BUILD_EXAMPLES)
  add_subdirectory(trac_ik_examples)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
